\begin{theorem}\leanok
[Right addition by zero]
  \label{thm:toolchain-demo}
  \lean{toolchain_demo}
  For every natural number $n$, one has $n+0=n$.
\end{theorem}
\begin{proof}

\leanok
  This is the right-identity law for addition on the natural numbers.
\end{proof}
